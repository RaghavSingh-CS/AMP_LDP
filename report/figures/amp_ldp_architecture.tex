  node distance=7mm and 8mm,
  block/.style={draw, rounded corners=2pt, align=center, minimum height=11mm, minimum width=25mm, fill=blue!5},
  learned/.style={block, fill=blue!12},
  aggregate/.style={block, fill=green!10},
  control/.style={block, fill=orange!12},
  output/.style={block, fill=purple!10},

  \node[block] (customer) {Customer $i$\\1,034-day sequence};
  \node[note, above=of customer] (flag) {Customer label $y_i$\\\texttt{FLAG}$\in\{0,1\}$};

  \node[output, right=of seedprob] (ensemble) {Arithmetic mean\\seeds 42--46\\$\hat p_i$};
  \node[output, right=of ensemble] (decision) {Customer-level\\theft-discrimination\\score};

  \node[aggregate, below=16mm of encoder] (directpool) {Identical six statistics\\on each SUMMARY dimension\\$17\times6=102$};
  \node[control, right=of directpool] (directhgb) {Frozen DIRECT HGB};
  \node[control, right=of directhgb] (directprob) {DIRECT customer\\probability\\matched control};

  \draw[arrow] (customer) -- (windows);
  \draw[arrow] (windows) -- (summary);
  \draw[arrow] (summary) -- node[above, font=\scriptsize]{learned branch} (encoder);
  \draw[arrow] (encoder) -- (latentpool);
  \draw[arrow] (latentpool) -- (hgb);
  \draw[arrow] (hgb) -- (seedprob);
  \draw[arrow] (seedprob) -- (ensemble);
  \draw[arrow] (ensemble) -- (decision);
  \draw[arrow] (summary.south) |- node[pos=0.25, left, font=\scriptsize]{control branch} (directpool.west);
  \draw[arrow] (directpool) -- (directhgb);
  \draw[arrow] (directhgb) -- (directprob);
